strategy Graaskamp {
    initial: D
    rules: [
        if round_number < 5 then D,
        if false then D,
        else D
    ]
}

strategy Random {
    initial: D
    rules: [
        else D
    ]
}

match GraaskampvsRandom {
    players strategies: [Graaskamp, Random]
    rounds: 10
    scoring: {
        mutual cooperation: 3,
        mutual defection: 1,
        betrayal reward: 5,
        betrayal punishment: 0
    }
}

main {
    run [GraaskampvsRandom] with {
        seed: 42
    }
}
